from flask import Flask, request, redirect, render_template_string import sqlite3 import os app = Flask(__name__) DB_NAME = "property.db" def get_connection(): conn = sqlite3.connect(DB_NAME) conn.row_factory = sqlite3.Row return conn def init_db(): conn = get_connection() conn.execute(""" CREATE TABLE IF NOT EXISTS properties ( id INTEGER PRIMARY KEY AUTOINCREMENT, transaction_type TEXT NOT NULL, property_type TEXT NOT NULL, area INTEGER NOT NULL, place TEXT NOT NULL, budget_range TEXT NOT NULL, property_value REAL NOT NULL, owner_name TEXT NOT NULL, contact TEXT NOT NULL ) """) conn.commit() conn.close() init_db() HTML = """ Chennai Property Hub

Chennai Property Buying & Rental SaaS

Property Listings

{% for row in properties %} {% endfor %}
ID Buy/Rent Type Area Place Range Value Owner Contact
{{ row["id"] }} {{ row["transaction_type"] }} {{ row["property_type"] }} {{ row["area"] }} {{ row["place"] }} {{ row["budget_range"] }} ₹ {{ row["property_value"] }} {{ row["owner_name"] }} {{ row["contact"] }}
""" @app.route("/") def home(): conn = get_connection() properties = conn.execute( "SELECT * FROM properties ORDER BY id DESC" ).fetchall() conn.close() return render_template_string( HTML, properties=properties ) @app.route("/add", methods=["POST"]) def add_property(): conn = get_connection() conn.execute( """ INSERT INTO properties ( transaction_type, property_type, area, place, budget_range, property_value, owner_name, contact ) VALUES (?, ?, ?, ?, ?, ?, ?, ?) """, ( request.form["transaction_type"], request.form["property_type"], int(request.form["area"]), request.form["place"], request.form["budget_range"], float(request.form["property_value"]), request.form["owner_name"], request.form["contact"] ) ) conn.commit() conn.close() return redirect("/") if __name__ == "__main__": app.run(host="0.0.0.0", port=5000, debug=True)
