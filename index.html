from flask import Flask, request, redirect, render_template_string import sqlite3 app = Flask(__name__) # Database Initialization def init_db(): conn = sqlite3.connect("property.db") cur = conn.cursor() cur.execute(""" CREATE TABLE IF NOT EXISTS properties ( id INTEGER PRIMARY KEY AUTOINCREMENT, transaction_type TEXT, property_type TEXT, area INTEGER, place TEXT, budget_range TEXT, property_value REAL, owner_name TEXT, contact TEXT ) """) conn.commit() conn.close() init_db() # Home Page @app.route("/") def home(): conn = sqlite3.connect("property.db") cur = conn.cursor() cur.execute("SELECT * FROM properties") properties = cur.fetchall() conn.close() html = """ Chennai Property Hub

Chennai Property Buying & Rental SaaS

Available Properties

{% for p in properties %} {% endfor %}
ID Buy/Rent Type Area Place Range Property Value Owner Contact
{{p[0]}} {{p[1]}} {{p[2]}} {{p[3]}} {{p[4]}} {{p[5]}} ₹ {{p[6]}} {{p[7]}} {{p[8]}}
""" return render_template_string( html, properties=properties ) # Add Property @app.route("/add", methods=["POST"]) def add_property(): transaction_type = request.form["transaction_type"] property_type = request.form["property_type"] area = request.form["area"] place = request.form["place"] budget_range = request.form["budget_range"] property_value = request.form["property_value"] owner_name = request.form["owner_name"] contact = request.form["contact"] conn = sqlite3.connect("property.db") cur = conn.cursor() cur.execute(""" INSERT INTO properties ( transaction_type, property_type, area, place, budget_range, property_value, owner_name, contact ) VALUES (?,?,?,?,?,?,?,?) """, ( transaction_type, property_type, area, place, budget_range, property_value, owner_name, contact )) conn.commit() conn.close() return redirect("/") if __name__ == "__main__": app.run(debug=True)
